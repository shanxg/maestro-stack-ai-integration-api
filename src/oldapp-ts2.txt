// 🔥 SECURITY REQUIREMENT: Load dotenv on the first line to read .env into API memory.
import 'dotenv/config';

// 1. INFRASTRUCTURE AND EXPRESS IMPORTS
import express, {type Request, type Response} from 'express';
import jwt from 'jsonwebtoken'; // 👈 Added to sign authentication tokens
import fs from 'node:fs'; // 👈 Added to read the .graphql file from disk
import path from 'node:path'; // 👈 Added to safely resolve directory paths
import http from 'node:http'; // 👈 Native Node module for managing network servers
import { UserRepository } from './repositories/UserRepository.js';
import { UserService } from './services/UserService.js';
import { UserController } from './controllers/UserController.js';
// 🔥 AI INTEGRATION
import { AIService } from './services/AIService.js';
import { AIController } from './controllers/AIController.js';

// 2. SECURITY PACKAGE IMPORTS (OWASP TOP 10)
import helmet from 'helmet';
import cors from 'cors';
import rateLimit from 'express-rate-limit';

// 3. MIDDLEWARE AND SCHEMA IMPORTS
import { authMiddleware } from './middlewares/authMiddleware.js';
import { validateMiddleware } from './middlewares/validateMiddleware.js';
import { createUserSchema } from './schemas/userSchema.js';

// 4. 🔥 GRAPHQL (APOLLO SERVER) IMPORTS
import { ApolloServer } from '@apollo/server';
import { expressMiddleware } from '@as-integrations/express5';
import { userResolvers } from './graphql/userResolvers.js';

// 5. 🔥 REAL-TIME ARCHITECTURE IMPORT
import { initializeSocket } from './graphql/socket.js'; // 👈 Added to load the WebSocket initializer

// 🔥 SAFE SECRET RETRIEVAL: Read the secret from the environment, with a fallback if it is empty.
const JWT_SECRET = process.env.JWT_SECRET || 'fallback_secret_key_backup_2026';

// 🔥 FACTORY FUNCTION:
// Wrap server assembly in an asynchronous function. If Jest passes a
// 'customUserService' mock, Express uses it to avoid connecting to real RabbitMQ/Redis.
export async function createApp(customUserService?: any) {

  const app = express();

  // 🔥 Create the unified Node HTTP server with Express attached.
  const server = http.createServer(app);
  // 🔥 Initialize the isolated WebSocket module with the unified HTTP server.
  initializeSocket(server);

  // ==========================================
  // 🔥 GLOBAL PROTECTION MIDDLEWARE
  // ==========================================

  // A. HELMET: Set robust HTTP headers to mitigate attacks such as clickjacking and XSS.
  // Disable CSP locally so the Apollo Sandbox can run.
  const helmetOptions =
    process.env.NODE_ENV === 'production'
      ? {}
      : { contentSecurityPolicy: false };
  app.use(helmet(helmetOptions));

  // B. CORS: Restrict which external web origins can access the API.
  app.use(cors({ 
    origin: 'http://localhost:3000', // Allow only the local frontend
    methods: ['GET', 'POST', 'PUT', 'DELETE'],
    allowedHeaders: ['Content-Type', 'Authorization']
  }));

  // C. RATE LIMIT: Protect the server from DDoS by limiting excessive requests per IP.
  const limiter = rateLimit({
    windowMs: 15 * 60 * 1000, // 15-minute counting window
    max: 100, // Allow at most 100 requests per IP during this window
    standardHeaders: true, // Add standard monitoring headers to the response
    legacyHeaders: false, // Disable legacy X-RateLimit-* headers
    message: {
      error: {
        code: 'TOO_MANY_REQUESTS',
        message: 'Você atingiu o limite máximo de requisições. Tente novamente em 15 minutos.'
      }
    }
  });

  // Apply brute-force protection to all Express routes.
  app.use(limiter);

  // Enable the native middleware for parsing JSON payloads.
  app.use(express.json());

  // ==========================================
  // 🏭 DEPENDENCY INJECTION (FACTORY PATTERN)
  // ==========================================
  // If a mock service is provided, inject it into the controller.
  // Otherwise, create the service connected to the live infrastructure.
  const userRepository = new UserRepository();
  const userService = customUserService || new UserService(userRepository);

  function makeUserController(): UserController {
    return new UserController(userService);
  }

  const userController = makeUserController();

  // =========================================================================
  // 🔐 AUTHENTICATION ENDPOINT: POST /auth/login
  // =========================================================================
  // Receive credentials, validate the identity, and issue a JWT.
  app.post('/auth/login', (req: Request, res: Response): void => {
    
    // 1. EXTRACT DATA: Read the username and password from the request body.
    const { username, password } = req.body;

    // 2. DATABASE MOCK: Check the tutorial's test credentials.
    // In production, retrieve the credentials from the database and verify them with bcrypt.
    if (username === 'admin' && password === 'secret123') {
      
      // 3. GENERATE JWT: Create and sign a token containing the logged-in user's payload.
      // - Argument 1: Public information to include in the token (username and role).
      // - Argument 2: The shared secret (JWT_SECRET), loaded securely from .env.
      // - Argument 3: Configuration that makes the token expire after one hour (TTL).
      const token = jwt.sign(
        { user: username, role: 'ADMIN' }, 
        JWT_SECRET, 
        { expiresIn: '1h' }
      );

      // 4. SUCCESS RESPONSE: Return the token as JSON so the client can store it in localStorage.
      res.json({ token });
      return;
    }

    // 5. AUTHENTICATION FAILURE: Reject invalid credentials with HTTP 401 (Unauthorized).
    // Return an enveloped error using the unified OWASP response format.
    res.status(401).json({
      error: {
        code: 'UNAUTHORIZED',
        message: 'Credenciais inválidas. Usuário ou senha incorretos.'
      }
    });
  });

  // ==========================================
  // 🛣️ MAESTRO REST ENDPOINT MAPPING
  // ==========================================
  // Pass explicit callbacks to preserve request context.
  app.get('/users', authMiddleware, (req: Request, res: Response) => userController.getUsers(req, res));       // Paginated user listing route
  app.get('/users/:id', authMiddleware, (req: Request, res: Response) => userController.getUserById(req, res)); // Find a user by ID
  app.post('/users', validateMiddleware(createUserSchema), (req: Request, res: Response) => userController.createUser(req, res));     // Enqueue a new user asynchronously

  // ==========================================
  // 🏭 AI INITIALIZATION (DAY 6)
  // ==========================================
  const aiService = new AIService();
  const aiController = new AIController(aiService);

  // 📝 POST /ai/chat is protected by JWT authentication.
  // authMiddleware rejects missing or invalid tokens with HTTP 401.
  // Otherwise, AIController validates the payload and sends the prompt to Llama 3.
  app.post('/ai/chat', authMiddleware, (req, res) => aiController.chat(req, res));


  // =========================================================================
  // 📡 🔥 REAL-TIME ENDPOINT: SERVER-SENT EVENTS (SSE) - GET /events
  // =========================================================================
  // One-way streaming channel for sending lightweight notifications directly to the browser.
  app.get('/events', (req: Request, res: Response): void => {
    // Set the HTTP headers required to keep the streaming connection open.
    res.setHeader('Content-Type', 'text/event-stream');
    res.setHeader('Cache-Control', 'no-cache');
    res.setHeader('Connection', 'keep-alive');

    // Declare the interval using the correct Node.js type.
    let interval: NodeJS.Timeout | undefined;

    // 🔥 DEFENSIVE CHECK: Start the heartbeat loop only outside the test environment.
    if (process.env.NODE_ENV !== 'test') {
      interval = setInterval(() => {
        res.write(`data: ${JSON.stringify({ timestamp: Date.now(), msg: 'Keep-alive Maestro' })}\n\n`);
      }, 5000);
    }

    // Listen for client disconnects and clear the interval to free server memory.
    req.on('close', () => {
      if (interval) {
        clearInterval(interval);
      }
      console.log('🔌 Conexão SSE de notificações encerrada pelo cliente.');
    });
  });


  // ==========================================
  // 🌌 🔥 GRAPHQL CONFIGURATION AND INITIALIZATION
  // ==========================================

  // Step A: Read schema.graphql and convert its contents to a string.
  const typeDefs = fs.readFileSync(
    path.resolve('src', 'graphql', 'schema.graphql'),
    'utf-8'
  );

  // Step B: Create Apollo Server with the type definitions and corresponding resolvers.
  const apolloServer = new ApolloServer({
    typeDefs,
    resolvers: userResolvers,
  });

 // 🔥 ESM CONCURRENCY FIX:
 // Start Apollo only outside the Jest test environment.
 // This prevents Express from hanging and avoids open handles.
 if (process.env.NODE_ENV !== 'test') {

    // Step C: Start Apollo before attaching it to the Express routes.
    await apolloServer.start();

    // Step D: Attach Apollo Server to the '/graphql' route using the native expressMiddleware.
    app.use('/graphql', expressMiddleware(apolloServer));
 } 
  
  // ==========================================
  // 🚀 SERVER STARTUP
  // ==========================================
  const PORT = Number(process.env.PORT) || 3000;
  // 🔥 Use server.listen instead of app.listen so WebSocket channels share the port.
  if (process.env.NODE_ENV === 'test') {
    console.log("⚠️ Modo de Testes Ativado: O servidor HTTP não será iniciado para evitar conflitos de porta.");
  }else {
    // 🌐 Bind to "0.0.0.0" so the Pod accepts connections from outside the container.
    server.listen(PORT, "0.0.0.0", () => {
        console.log(`🚀 API REST, GraphQL & WebSockets rodando com segurança em http://localhost:${PORT}`);
        console.log(`🌌 Sandbox do GraphQL ativo em http://localhost:${PORT}/graphql`);
        console.log(`📡 Canal de notificações SSE ativado em http://localhost:${PORT}/events`);
    });
  }

  return server; // Return the server for tests and external use.
}
