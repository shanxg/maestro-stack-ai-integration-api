// 1. IMPORTS: Bring in the Express framework and all classes created for this project.
import express from 'express';
import { UserRepository } from './repositories/UserRepository.js';
import { UserService } from './services/UserService.js';
import { UserController } from './controllers/UserController.js';

// 2. INITIAL CONFIGURATION: Initialize Express and enable middleware for reading JSON data.
const app = express();
app.use(express.json());

// === APPLY THE FACTORY DESIGN PATTERN ===
// This function is a "Controller Factory." Instead of spreading object creation across the code,
// centralize it here. It knows how to build a ready-to-use UserController.
function makeUserController(): UserController {
  
  // Step A: Create the in-memory repository.
  const userRepository = new UserRepository();
  
  // Step B: Create the service and inject the repository.
  const userService = new UserService(userRepository);
  
  // Step C: Create the controller, inject the service, and return it to the caller.
  return new UserController(userService);
}

// Use the factory to create the controller that handles API routes.
const userController = makeUserController();


// === ROUTE MAPPING ===
// Connect URL paths to methods on the controller instance.
app.get('/users', userController.getUsers);       // Route to list all users
app.get('/users/:id', userController.getUserById); // Route to find a user by ID
app.post('/users', userController.createUser);     // Route to create a user


// === SERVER STARTUP ===
// Configure port 3000 and start the API to receive requests.
const PORT = 3000;
app.listen(PORT, () => {
  console.log(`🚀 API rodando com sucesso no endereço http://localhost:${PORT}`);
});
